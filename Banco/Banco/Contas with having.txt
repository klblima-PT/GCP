
WITH conta as
(
select distinct codagencia, nroconta,cgc_cpfendereco, databertura, codgerente, codsituacao, restricao, dt_carga_bronze, count(*) from `corporativo-poc-hvar.bronze_bi_rendimento.bi_contacorrente_tbl_contacorrente_conta`
group by codagencia, nroconta,cgc_cpfendereco, databertura, codgerente, codsituacao, restricao, dt_carga_bronze
having count(*) > 1
), 
movto as
(
select distinct codagencia, nroconta, datamovto, nromovimento, natureza,bcoorigem, ageorigem, origem, valor,tarifado
,debitado, entradaage, dt_carga_bronze, count(*) 
from `corporativo-poc-hvar.bronze_bi_rendimento.bi_contacorrente_tbl_contacorrentemovto` 
group by codagencia, nroconta, datamovto, nromovimento, natureza,bcoorigem, ageorigem, origem, valor,tarifado, debitado, entradaage, dt_carga_bronze
having count(*) > 1
),
cliente as 
(
select distinct codcliente, cgc_cpf , nome, tipopessoa, codgerente,numdocumento,tipodocumento, orgaoexpedidor,uf_orgaoexpedidor, dt_carga_bronze 
, count(*)   from `corporativo-poc-hvar.bronze_bi_rendimento.bi_infobank_tbl_infobanc_clientes`
 group by codcliente, cgc_cpf , nome, tipopessoa, codgerente,numdocumento,tipodocumento, orgaoexpedidor,uf_orgaoexpedidor, dt_carga_bronze
 having count(*) > 1
),
pessoa as
(
select distinct cgc_cpf, nome, nomefantasia, tipopessoa, status,numdocumento, tipodocumento, orgaoexpedidor, dt_carga_bronze , count(*) from `corporativo-poc-hvar.bronze_bi_rendimento.bi_infobank_tbl_infobanc_pessoas` 
group by cgc_cpf, nome, nomefantasia, tipopessoa, status,numdocumento, tipodocumento, orgaoexpedidor, dt_carga_bronze
having count(*) > 1
)

select * from conta a 
left join movto b
on a.codagencia = b.codagencia
and  a.nroconta = b.nroconta
